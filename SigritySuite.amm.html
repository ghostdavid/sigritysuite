open_library(...)
open_library(fileName: str, external: bool = False) -> None
 
Load analysis models from the specified AMM library file.
 
Opens and imports device models (VRM, IC, inductor, resistor, etc.) from an Analysis Model Manager (AMM) library file into either the project library or external library. Use this to make pre-defined component models available for assignment in your design.
 
Parameters
  fileName: str - Full path to the AMM library file to load.
  external: bool = False - If True, load into the external library; if False, load into the project library.
 
Returns
  None
 
Example
  import amm
 
  mgr = amm.get_manager()
  # Load into project library
  mgr.open_library('C:/models/vrm_models.amm')
  # Load into external library
  mgr.open_library('C:/models/external_models.amm', True)

search_models(...)
search_models(model_type: str, properties: dict, lib_name: str = 'all') -> list[dict]
 
Search loaded AMM libraries for models matching the given property filters.
 
Each entry in the result list is a dict with keys:
  'model_name' : str  - The Part No. / model identifier in the library.
  'lib'        : str  - 'project' for the project library, or the file path for external libs.
 
Parameters
  model_type : str  - Model category. One of: IC, capacitor, inductor, resistor, VRM, connector, package, Spice.
  properties : dict - Property filters. Keys are property names; values are either a plain string/number
                      (uses '=' operator) or a 2-tuple (operator, value) where operator is one of =, <, <=, >, >=.
                      All filters are AND-combined. Value strings accept unit suffixes (e.g. '100n' for 100 nF).
                      Capacitor properties: capSize, capCNOM, capVoltage, capTCC, capPref,
                        capCompCost, capMountCost, capBomCost, capArea, upperTol, lowerTol, comment.
                      Common properties (all types): modelName, manufacturer, mfgPartNo, lookupKeyword.
  lib_name   : str  - Library scope (default 'all'):
                        'all'         - search project library first, then external.
                        'project'     - project library only.
                        'externalAll' - all external lib bins.
                        '<path>'      - a specific external lib bin by name or path.
 
Returns
  list[dict]  - Matching models, each {'model_name': str, 'lib': str}. Empty list if none found.
 
Example
  import amm
 
  mgr = amm.get_manager()
  mgr.open_library('C:/models/caps.amm', True)
 
  # Find all 0402E / 100 nF capacitors rated >= 6.3 V with X7R dielectric
  results = mgr.search_models(
      'capacitor',
      {'capSize': '0402E', 'capCNOM': '100n', 'capVoltage': ('>=', '6.3'), 'capTCC': 'X7R'})
 
  for r in results:
      print(r['model_name'], r['lib'])

get_manager(...)
get_manager() -> amm.AnalysisModelManager
 
Return the singleton Analysis Model Manager (AMM) instance.
 
Retrieves the global manager object used to load, organize, and access device analysis models (VRM, IC, inductor, resistor, capacitor, etc.) from AMM library files. Use this to access the AMM API for managing component models in your project.
 
Returns
  amm.AnalysisModelManager - The singleton AMM manager instance.
 
Example
  import amm
 
  # Get the AMM manager instance
  mgr = amm.get_manager()
  # Load models from library file
  mgr.open_library('C:/models/vrm_models.amm')
